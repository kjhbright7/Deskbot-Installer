drawVLine(cx - x, cy - y, 2 * y + 1, color); drawVLine(cx + y, cy - x, 2 * x + 1, color); drawVLine(cx - y, cy - x, 2 * x + 1, color); } } function drawVLine(x, y, len, color) { for (let i = 0; i < len; i++) { let py = y + i; if (x >= 0 && x < OLED_W && py >= 0 && py < OLED_H) { oledGrid[py][x] = color; } } } // ★ OLED 실물 사진과 100% 똑같이 나오는 귀여운 곡선 냥이 입 (ㅅ) function drawCatMouth(cx, cy) { // 왼쪽 호 (Lower Right) const leftArc = [ {dx: -5, dy: 0}, {dx: -4, dy: 1}, {dx: -3, dy: 2}, {dx: -2, dy: 3}, {dx: -1, dy: 4}, {dx: 0, dy: 4} ]; }); } // 케이스 베젤에 절대 가려지지 않는 안전 영역 볼터치 (두께감 있는 빗금) function drawSafeBlush(cx, cy, isRight = false) { for (let i = 0; i < 3; i++) { let ox = cx + (i * 3); let oy = cy; if (isRight) { // 오른쪽 볼터치 (///) setPixel(ox, oy);
